cmake_minimum_required(VERSION 3.20)
project(visual_program_experiments LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-fno-exceptions -fno-rtti -Wall -Wextra -Wpedantic -Werror -Wold-style-cast)

# Use Cargo's complete bundled compiler, not a mixture of system versions.
set(SHADERC_ARCHIVE "" CACHE FILEPATH "Cargo-built libshaderc_combined.a")
if(NOT EXISTS "${SHADERC_ARCHIVE}" OR NOT SHADERC_ARCHIVE MATCHES "\\.a$")
  message(FATAL_ERROR "Set SHADERC_ARCHIVE to Cargo's shaderc-sys OUT_DIR/lib/libshaderc_combined.a")
endif()
add_library(visual_compiler STATIC ../../src/shader_compiler.cpp)
target_include_directories(visual_compiler PUBLIC ../../src)
target_link_libraries(visual_compiler PUBLIC "${SHADERC_ARCHIVE}" pthread)
add_executable(compiler_probe compiler_probe.cpp)
target_link_libraries(compiler_probe PRIVATE visual_compiler)

# The experiment reuses the existing context's fixed baseline pipeline; it needs
# the Cargo-generated header, but its agent programs compile in memory.
set(BASELINE_HEADER_DIR "" CACHE PATH "Cargo OUT_DIR containing visual_spirv.h")
if(NOT EXISTS "${BASELINE_HEADER_DIR}/visual_spirv.h")
  message(FATAL_ERROR "Set BASELINE_HEADER_DIR to an existing Cargo-generated shader header directory")
endif()
add_library(visual_native STATIC ../../src/visual_gpu.cpp)
target_include_directories(visual_native PUBLIC ../../src "${BASELINE_HEADER_DIR}")
target_link_libraries(visual_native PUBLIC visual_compiler vulkan)
add_executable(gpu_probe gpu_probe.cpp)
target_include_directories(gpu_probe PRIVATE "${BASELINE_HEADER_DIR}")
target_link_libraries(gpu_probe PRIVATE visual_native)
add_executable(normalization_probe normalization_probe.cpp)
target_include_directories(normalization_probe PRIVATE "${BASELINE_HEADER_DIR}")
target_link_libraries(normalization_probe PRIVATE visual_native)
add_executable(runtime_probe runtime_probe.cpp)
target_include_directories(runtime_probe PRIVATE "${BASELINE_HEADER_DIR}")
target_link_libraries(runtime_probe PRIVATE visual_native)
enable_testing()
add_test(NAME in_memory_compiler COMMAND compiler_probe)
add_executable(runtime_edge_probe runtime_edge_probe.cpp)
target_include_directories(runtime_edge_probe PRIVATE "${BASELINE_HEADER_DIR}")
target_link_libraries(runtime_edge_probe PRIVATE visual_native)
add_executable(concurrency_probe concurrency_probe.cpp)
target_include_directories(concurrency_probe PRIVATE "${BASELINE_HEADER_DIR}")
target_link_libraries(concurrency_probe PRIVATE visual_native)
